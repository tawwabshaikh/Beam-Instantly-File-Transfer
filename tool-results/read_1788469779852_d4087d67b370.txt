    41→export function MobileSession({ code, token }: { code: string; token: string }) {
    42→  const phase = useBeamStore((s) => s.phase)
    43→  const session = useBeamStore((s) => s.session)
    44→  const openSession = useBeamStore((s) => s.openSession)
    45→  const resetAll = useBeamStore((s) => s.resetAll)
    46→
    47→  useEffect(() => {
    48→    void openSession(code, token)
    49→    return () => resetAll()
    50→  }, [code, token, openSession, resetAll])
    51→
    52→  return (
    53→    <div className="flex min-h-screen flex-col bg-background">
    54→      <MobileHeader code={code} expiresAt={session?.expiresAt ?? 0} />
    55→
    56→      <main className="mx-auto w-full max-w-md flex-1 px-4 pb-10 pt-5 safe-bottom">
    57→        {phase === 'opening' && <OpeningView />}
    58→        {phase === 'connecting' && <ConnectingView />}
    59→        {(phase === 'connected' ||
    60→          phase === 'waiting' ||
    61→          phase === 'creating' ||
    62→          phase === 'idle') && <ConnectedView />}
    63→        {phase === 'lost' && <LostView />}
    64→        {phase === 'expired' && <ExpiredView />}
    65→        {phase === 'invalid' && <InvalidView />}
    66→        {phase === 'ended' && <EndedView />}
    67→        {phase === 'failed' && <FailedView />}
    68→      </main>
    69→
    70→      <footer className="border-t border-border/60 px-4 py-3 text-center text-[11px] text-muted-foreground safe-bottom">
    71→        <span className="inline-flex items-center gap-1">
    72→          <ShieldCheck className="h-3 w-3 text-primary" aria-hidden />
    73→          Encrypted session · Files go device-to-device
    74→        </span>
    75→      </footer>
    76→    </div>
    77→  )
    78→}

   317→/** Aggregate progress for a batch of downloads (e.g. "Download All"). */
   318→function BatchProgress({ rows }: { rows: TransferRow[] }) {
   319→  const relevant = rows.filter((r) => r.status !== 'canceled')
   320→  const total = relevant.reduce((a, r) => a + r.size, 0)
   321→  const done = relevant.reduce((a, r) => a + r.transferred, 0)
   322→  const active = relevant.filter((r) => r.status === 'active' || r.status === 'queued').length
   323→  const finished = relevant.filter((r) => r.status === 'done').length
   324→  const pct = total > 0 ? Math.min(100, (done / total) * 100) : 0
   325→
   326→  if (relevant.length < 2) return null
   327→
   328→  return (
   329→    <div className="bg-muted/40 px-4 py-2.5">
   330→      <div className="flex items-center justify-between text-xs">
   331→        <span className="font-medium text-foreground">
   332→          {active > 0 ? `Downloading ${finished + 1} of ${relevant.length}` : `${finished} of ${relevant.length} downloaded`}
   333→        </span>
   334→        <span className="tnum text-muted-foreground">
   335→          {formatBytes(done)} / {formatBytes(total)}
   336→        </span>
   337→      </div>
   338→      <Progress value={pct} className="mt-1.5 h-1.5" aria-label={`Batch progress ${pct.toFixed(0)}%`} />
   339→    </div>
   340→  )
   341→}

   343→function MobileFileRow({ fileId, flash = false }: { fileId: string; flash?: boolean }) {
   344→  const file = useBeamStore((s) => s.manifest.find((f) => f.id === fileId))
   345→  const transfers = useBeamStore((s) => s.transfers)
   346→  const requestDownload = useBeamStore((s) => s.requestDownload)
   347→  const shareReceived = useBeamStore((s) => s.shareReceived)
   348→  const retryTransfer = useBeamStore((s) => s.retryTransfer)
   349→
   350→  const rows = useMemo(
   351→    () =>
   352→      Object.values(transfers).filter((t) => t.fileId === fileId && t.direction === 'd2p'),
   353→    [transfers, fileId],
   354→  )
   355→
   356→  if (!file) return null
   357→  const row = rows[rows.length - 1]
   358→  const busy = row && (row.status === 'queued' || row.status === 'active')
   359→  const failed = row?.status === 'error'
   360→  const done = rows.some((r) => r.status === 'done')
   361→
   362→  return (
   363→    <li className={cn('flex items-center gap-3 px-4 py-3 transition-colors hover:bg-accent/40', flash && 'animate-flash')}>
   364→      <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-muted">
   365→        <FileTypeIcon type={file.type} name={file.name} className="h-4.5 w-4.5" />
   366→      </span>
   367→      <div className="min-w-0 flex-1">
   368→        <p className="truncate text-sm font-medium" title={file.name}>
   369→          {file.name}
   370→        </p>
   371→        <p className={cn('tnum truncate text-xs', failed ? 'text-destructive' : 'text-muted-foreground')}>
   372→          {row?.status === 'active'
   373→            ? `${(row.size > 0 ? (row.transferred / row.size) * 100 : 0).toFixed(0)}% · ${formatSpeed(row.speed)}`
   374→            : failed
   375→              ? (row?.error ?? 'Download failed')
   376→              : formatBytes(file.size)}
   377→        </p>
   378→      </div>
   379→      {done ? (
   380→        <div className="flex items-center gap-1">
   381→          <Button
   382→            variant="ghost"
   383→            size="icon"
   384→            className="h-8 w-8"
   385→            onClick={() => {
   386→              const rec = useBeamStore.getState().received.find((r) => r.name === file.name)
   387→              if (rec) void shareReceived(rec.id)
   388→            }}
   389→            aria-label={`Share or re-save ${file.name}`}
   390→          >
   391→            <Share2 className="h-4 w-4" />
   392→          </Button>
   393→          <span className="inline-flex items-center gap-1 text-xs font-medium text-primary">
   394→            <CheckCircle2 className="h-4 w-4" aria-hidden />
   395→            Saved
   396→          </span>
   397→        </div>
   398→      ) : failed ? (
   399→        <Button
   400→          size="sm"
   401→          variant="outline"
   402→          className="h-9 min-w-20 border-destructive/40 text-destructive hover:bg-destructive/10 hover:text-destructive"
   403→          onClick={() => retryTransfer(row!.id)}
   404→        >
   405→          <RotateCcw className="h-4 w-4" aria-hidden />
   406→          Retry
   407→        </Button>
   408→      ) : (
   409→        <Button size="sm" className="h-9 min-w-24" onClick={() => requestDownload(fileId)} disabled={busy}>
   410→          {busy ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Download className="h-4 w-4" aria-hidden />}
   411→          {busy ? 'Busy…' : 'Download'}
   412→        </Button>
   413→      )}
   414→    </li>
   415→  )
   416→}

   528→              </li>
   529→            )
   530→          })}
   531→        </ul>
   532→      )}
   533→
   534→      {active && (

   583→function InvalidView() {
   584→  const error = useBeamStore((s) => s.error)
   585→  const [manual, setManual] = useState('')
   586→  const openSession = useBeamStore((s) => s.openSession)
   587→
   588→  const tryManual = () => {
   589→    const v = manual.trim()
   590→    try {
   591→      const url = new URL(v)
   592→      const s = url.searchParams.get('s')
   593→      const t = url.searchParams.get('t')
   594→      if (s && t) {
   595→        window.location.assign(`/?s=${encodeURIComponent(s)}&t=${encodeURIComponent(t)}`)
   596→        return
   597→      }
   598→    } catch {
   599→      // not a URL
   600→    }
   601→    const m = v.match(/^([A-HJ-NP-Z2-9]{6})-([A-Za-z0-9]{8,128})$/i)
   602→    if (m) {
   603→      window.location.assign(`/?s=${encodeURIComponent(m[1]!.toUpperCase())}&t=${encodeURIComponent(m[2]!)}`)
   604→      return
   605→    }
   606→    window.location.assign('/')
   607→  }
   608→
   609→  return (
   610→    <StateShell tone="border-destructive/30 bg-destructive/5">
   611→      <span className="flex h-14 w-14 items-center justify-center rounded-full bg-destructive/10">
   612→        <CloudOff className="h-6 w-6 text-destructive" aria-hidden />
   613→      </span>
   614→      <p className="mt-4 font-semibold">{error?.title ?? 'Invalid or expired link'}</p>
   615→      <p className="mt-1 text-sm text-muted-foreground">
   616→        {error?.message ?? 'This QR link is no longer valid. Scan a fresh code from the desktop.'}
   617→      </p>
   618→      <div className="mt-5 w-full space-y-2">
   619→        <Input
   620→          value={manual}
   621→          onChange={(e) => setManual(e.target.value)}
   622→          placeholder="Paste pairing link or CODE-KEY"
   623→          aria-label="Pairing link or manual code"
   624→          className="h-11 bg-background text-center"
   625→        />
   626→        <Button className="h-11 w-full" onClick={tryManual}>
   627→          <FilePlus2 className="h-4 w-4" aria-hidden />
   628→          Try pairing key
   629→        </Button>
   630→      </div>
   631→      <StartOverLink />
   632→    </StateShell>
   633→  )
   634→}
