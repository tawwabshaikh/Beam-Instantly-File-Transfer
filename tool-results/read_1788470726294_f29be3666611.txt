   318→   318→/** Aggregate progress for a batch of downloads (e.g. "Download All"). */
   319→   319→function BatchProgress({ rows }: { rows: TransferRow[] }) {
   320→   320→  const relevant = rows.filter((r) => r.status !== 'canceled')
   321→   321→  const total = relevant.reduce((a, r) => a + r.size, 0)
   322→   322→  const done = relevant.reduce((a, r) => a + r.transferred, 0)
   323→   323→  const active = relevant.filter((r) => r.status === 'active' || r.status === 'queued').length
   324→   324→  const finished = relevant.filter((r) => r.status === 'done').length
   325→   325→  const pct = total > 0 ? Math.min(100, (done / total) * 100) : 0
   326→   326→
   327→   327→  if (relevant.length < 2) return null
   328→   328→
   329→   329→  return (
   330→   330→    <div className="bg-muted/40 px-4 py-2.5">
   331→   331→      <div className="flex items-center justify-between text-xs">
   332→   332→        <span className="font-medium text-foreground">
   333→   333→          {active > 0 ? `Downloading ${finished + 1} of ${relevant.length}` : `${finished} of ${relevant.length} downloaded`}
   334→   334→        </span>
   335→   335→        <span className="tnum text-muted-foreground">
   336→   336→          {formatBytes(done)} / {formatBytes(total)}
   337→   337→        </span>
   338→   338→      </div>
   339→   339→      <Progress value={pct} className="mt-1.5 h-1.5" aria-label={`Batch progress ${pct.toFixed(0)}%`} />
   340→   340→    </div>
   341→   341→  )
   342→   342→}

   344→   344→function MobileFileRow({ fileId, flash = false }: { fileId: string; flash?: boolean }) {
   345→   345→  const file = useBeamStore((s) => s.manifest.find((f) => f.id === fileId))
   346→   346→  const transfers = useBeamStore((s) => s.transfers)
   347→   347→  const requestDownload = useBeamStore((s) => s.requestDownload)
   348→   348→  const shareReceived = useBeamStore((s) => s.shareReceived)
   349→   349→  const retryTransfer = useBeamStore((s) => s.retryTransfer)
   350→   350→
   351→   351→  const rows = useMemo(
   352→   352→    () =>
   353→   353→      Object.values(transfers).filter((t) => t.fileId === fileId && t.direction === 'd2p'),
   354→   354→    [transfers, fileId],
   355→   355→  )
   356→   356→
   357→   357→  if (!file) return null
   358→   358→  const row = rows[rows.length - 1]
   359→   359→  const busy = row && (row.status === 'queued' || row.status === 'active')
   360→   360→  const failed = row?.status === 'error'
   361→   361→  const done = rows.some((r) => r.status === 'done')
   362→   362→  // Once downloaded, images get a real thumbnail from the received blob.
   363→   363→  const receivedThumb = done && file.type.startsWith('image/')
   364→   364→    ? useBeamStore.getState().received.find((r) => r.name === file.name && r.size === file.size)?.url ?? null
   365→   365→    : null
   366→   366→
   367→   367→  return (
   368→   368→    <li className={cn('flex items-center gap-3 px-4 py-3 transition-colors hover:bg-accent/40', flash && 'animate-flash')}>
   369→   369→      {receivedThumb ? (
   370→   370→        <img
   371→   371→          src={receivedThumb}
   372→   372→          alt=""
   373→   373→          className="h-10 w-10 shrink-0 rounded-lg border border-border object-cover"
   374→   374→          loading="lazy"
   375→   375→        />
   376→   376→      ) : (
   377→   377→        <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-muted">
   378→   378→          <FileTypeIcon type={file.type} name={file.name} className="h-4.5 w-4.5" />
   379→   379→        </span>
   380→   380→      )}
   381→   381→      <div className="min-w-0 flex-1">
   382→   382→        <p className="truncate text-sm font-medium" title={file.name}>
   383→   383→          {file.name}
   384→   384→        </p>
   385→   385→        <p className={cn('tnum truncate text-xs', failed ? 'text-destructive' : 'text-muted-foreground')}>
   386→   386→          {row?.status === 'active'
   387→   387→            ? `${(row.size > 0 ? (row.transferred / row.size) * 100 : 0).toFixed(0)}% · ${formatSpeed(row.speed)}`
   388→   388→            : failed
   389→   389→              ? (row?.error ?? 'Download failed')
   390→   390→              : formatBytes(file.size)}
   391→   391→        </p>
   392→   392→      </div>
   393→   393→      {done ? (
   394→   394→        <div className="flex items-center gap-1">
   395→   395→          <Button
   396→   396→            variant="ghost"
   397→   397→            size="icon"
   398→   398→            className="h-8 w-8"
   399→   399→            onClick={() => {
   400→   400→              const rec = useBeamStore.getState().received.find((r) => r.name === file.name)
   401→   401→              if (rec) void shareReceived(rec.id)
   402→   402→            }}
   403→   403→            aria-label={`Share or re-save ${file.name}`}
   404→   404→          >
   405→   405→            <Share2 className="h-4 w-4" />
   406→   406→          </Button>
   407→   407→          <span className="inline-flex items-center gap-1 text-xs font-medium text-primary">
   408→   408→            <CheckCircle2 className="h-4 w-4" aria-hidden />
   409→   409→            Saved
   410→   410→          </span>
   411→   411→        </div>
   412→   412→      ) : failed ? (
   413→   413→        <Button
   414→   414→          size="sm"
   415→   415→          variant="outline"
   416→   416→          className="h-9 min-w-20 border-destructive/40 text-destructive hover:bg-destructive/10 hover:text-destructive"
   417→   417→          onClick={() => retryTransfer(row!.id)}
   418→   418→        >
   419→   419→          <RotateCcw className="h-4 w-4" aria-hidden />
   420→   420→          Retry
   421→   421→        </Button>
   422→   422→      ) : (
   423→   423→        <Button size="sm" className="h-9 min-w-24" onClick={() => requestDownload(fileId)} disabled={busy}>

   606→   606→function InvalidView() {
   607→   607→  const error = useBeamStore((s) => s.error)
   608→   608→  const [manual, setManual] = useState('')
   609→   609→  const openSession = useBeamStore((s) => s.openSession)
   610→   610→
   611→   611→  const tryManual = () => {
   612→   612→    const v = manual.trim()
   613→   613→    try {
   614→   614→      const url = new URL(v)
   615→   615→      const s = url.searchParams.get('s')
   616→   616→      const t = url.searchParams.get('t')
   617→   617→      if (s && t) {
   618→   618→        window.location.assign(`/?s=${encodeURIComponent(s)}&t=${encodeURIComponent(t)}`)
   619→   619→        return
   620→   620→      }
   621→   621→    } catch {
   622→   622→      // not a URL
   623→   623→    }
   624→   624→    const m = v.match(/^([A-HJ-NP-Z2-9]{6})-([A-Za-z0-9]{8,128})$/i)
   625→   625→    if (m) {
   626→   626→      window.location.assign(`/?s=${encodeURIComponent(m[1]!.toUpperCase())}&t=${encodeURIComponent(m[2]!)}`)
   627→   627→      return
   628→   628→    }
   629→   629→    window.location.assign('/')
   630→   630→  }
   631→   631→
   632→   632→  return (
   633→   633→    <StateShell tone="border-destructive/30 bg-destructive/5">
   634→   634→      <span className="flex h-14 w-14 items-center justify-center rounded-full bg-destructive/10">
   635→   635→        <CloudOff className="h-6 w-6 text-destructive" aria-hidden />
   636→   636→      </span>
   637→   637→      <p className="mt-4 font-semibold">{error?.title ?? 'Invalid or expired link'}</p>
   638→   638→      <p className="mt-1 text-sm text-muted-foreground">
   639→   639→        {error?.message ?? 'This QR link is no longer valid. Scan a fresh code from the desktop.'}
   640→   640→      </p>
   641→   641→      <div className="mt-5 w-full space-y-2">
   642→   642→        <Input
   643→   643→          value={manual}
   644→   644→          onChange={(e) => setManual(e.target.value)}
   645→   645→          placeholder="Paste pairing link or CODE-KEY"
   646→   646→          aria-label="Pairing link or manual code"
