    36→/**
    37→ * Phone experience — opened by scanning the desktop's QR code.
    38→ * Single-screen, thumb-friendly, all 11 lifecycle states.
    39→ */
    40→export function MobileSession({ code, token }: { code: string; token: string }) {
    41→  const phase = useBeamStore((s) => s.phase)
    42→  const session = useBeamStore((s) => s.session)
    43→  const openSession = useBeamStore((s) => s.openSession)
    44→  const resetAll = useBeamStore((s) => s.resetAll)
    45→
    46→  useEffect(() => {
    47→    void openSession(code, token)
    48→    return () => resetAll()
    49→  }, [code, token, openSession, resetAll])
    50→
    51→  return (
    52→    <div className="flex min-h-screen flex-col bg-background">
    53→      <MobileHeader code={code} expiresAt={session?.expiresAt ?? 0} />
    54→
    55→      <main className="mx-auto w-full max-w-md flex-1 px-4 pb-10 pt-5 safe-bottom">
    56→        {phase === 'opening' && <OpeningView />}
    57→        {phase === 'connecting' && <ConnectingView />}
    58→        {(phase === 'connected' ||
    59→          phase === 'waiting' ||
    60→          phase === 'creating' ||
    61→          phase === 'idle') && <ConnectedView />}
    62→        {phase === 'lost' && <LostView />}
    63→        {phase === 'expired' && <ExpiredView />}
    64→        {phase === 'invalid' && <InvalidView />}
    65→        {phase === 'ended' && <EndedView />}
    66→        {phase === 'failed' && <FailedView />}
    67→      </main>
    68→
    69→      <footer className="border-t border-border/60 px-4 py-3 text-center text-[11px] text-muted-foreground safe-bottom">
    70→        <span className="inline-flex items-center gap-1">
    71→          <ShieldCheck className="h-3 w-3 text-primary" aria-hidden />
    72→          Encrypted session · Files go device-to-device
    73→        </span>
    74→      </footer>
    75→    </div>
    76→  )
    77→}

   313→/** Aggregate progress for a batch of downloads (e.g. "Download All"). */
   314→function BatchProgress({ rows }: { rows: TransferRow[] }) {
   315→  const relevant = rows.filter((r) => r.status !== 'canceled')
   316→  const total = relevant.reduce((a, r) => a + r.size, 0)
   317→  const done = relevant.reduce((a, r) => a + r.transferred, 0)
   318→  const active = relevant.filter((r) => r.status === 'active' || r.status === 'queued').length
   319→  const finished = relevant.filter((r) => r.status === 'done').length
   320→  const pct = total > 0 ? Math.min(100, (done / total) * 100) : 0
   321→
   322→  if (relevant.length < 2) return null
   323→
   324→  return (
   325→    <div className="bg-muted/40 px-4 py-2.5">
   326→      <div className="flex items-center justify-between text-xs">
   327→        <span className="font-medium text-foreground">
   328→          {active > 0 ? `Downloading ${finished + 1} of ${relevant.length}` : `${finished} of ${relevant.length} downloaded`}
   329→        </span>
   330→        <span className="tnum text-muted-foreground">
   331→          {formatBytes(done)} / {formatBytes(total)}
   332→        </span>
   333→      </div>
   334→      <Progress value={pct} className="mt-1.5 h-1.5" aria-label={`Batch progress ${pct.toFixed(0)}%`} />
   335→    </div>
   336→  )
   337→}

   339→function MobileFileRow({ fileId, flash = false }: { fileId: string; flash?: boolean }) {
   340→  const file = useBeamStore((s) => s.manifest.find((f) => f.id === fileId))
   341→  const transfers = useBeamStore((s) => s.transfers)
   342→  const requestDownload = useBeamStore((s) => s.requestDownload)
   343→  const shareReceived = useBeamStore((s) => s.shareReceived)
   344→  const retryTransfer = useBeamStore((s) => s.retryTransfer)
   345→
   346→  const rows = useMemo(
   347→    () =>
   348→      Object.values(transfers).filter((t) => t.fileId === fileId && t.direction === 'd2p'),
   349→    [transfers, fileId],
   350→  )
   351→
   352→  if (!file) return null
   353→  const row = rows[rows.length - 1]
   354→  const busy = row && (row.status === 'queued' || row.status === 'active')
   355→  const failed = row?.status === 'error'
   356→  const done = rows.some((r) => r.status === 'done')
   357→
   358→  return (
   359→    <li className={cn('flex items-center gap-3 px-4 py-3 transition-colors hover:bg-accent/40', flash && 'animate-flash')}>
   360→      <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-muted">
   361→        <FileTypeIcon type={file.type} name={file.name} className="h-4.5 w-4.5" />
   362→      </span>
   363→      <div className="min-w-0 flex-1">
   364→        <p className="truncate text-sm font-medium" title={file.name}>
   365→          {file.name}
   366→        </p>
   367→        <p className={cn('tnum truncate text-xs', failed ? 'text-destructive' : 'text-muted-foreground')}>
   368→          {row?.status === 'active'
   369→            ? `${(row.size > 0 ? (row.transferred / row.size) * 100 : 0).toFixed(0)}% · ${formatSpeed(row.speed)}`
   370→            : failed
   371→              ? (row?.error ?? 'Download failed')
   372→              : formatBytes(file.size)}
   373→        </p>
   374→      </div>
   375→      {done ? (
   376→        <div className="flex items-center gap-1">
   377→          <Button
   378→            variant="ghost"
   379→            size="icon"
   380→            className="h-8 w-8"
   381→            onClick={() => {
   382→              const rec = useBeamStore.getState().received.find((r) => r.name === file.name)
   383→              if (rec) void shareReceived(rec.id)
   384→            }}
   385→            aria-label={`Share or re-save ${file.name}`}
   386→          >
   387→            <Share2 className="h-4 w-4" />
   388→          </Button>
   389→          <span className="inline-flex items-center gap-1 text-xs font-medium text-primary">
   390→            <CheckCircle2 className="h-4 w-4" aria-hidden />
   391→            Saved
   392→          </span>
   393→        </div>
   394→      ) : failed ? (
   395→        <Button
   396→          size="sm"
   397→          variant="outline"
   398→          className="h-9 min-w-20 border-destructive/40 text-destructive hover:bg-destructive/10 hover:text-destructive"
   399→          onClick={() => retryTransfer(row!.id)}
   400→        >
   401→          <RotateCcw className="h-4 w-4" aria-hidden />
   402→          Retry
   403→        </Button>
   404→      ) : (
   405→        <Button size="sm" className="h-9 min-w-24" onClick={() => requestDownload(fileId)} disabled={busy}>
   406→          {busy ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Download className="h-4 w-4" aria-hidden />}
   407→          {busy ? 'Busy…' : 'Download'}
   408→        </Button>
   409→      )}
   410→    </li>
   411→  )
   412→}

   511→                  </Button>

   514→                    variant="ghost"

   520→                  >
   521→                    <X className="h-4 w-4" />
   522→                  </Button>

   524→              </li>
   525→            )

   527→        </ul>
   528→      )}
   529→
   530→      {active && (

   579→function InvalidView() {
   580→  const error = useBeamStore((s) => s.error)
   581→  const [manual, setManual] = useState('')
   582→  const openSession = useBeamStore((s) => s.openSession)
   583→
   584→  const tryManual = () => {
   585→    const v = manual.trim()
   586→    try {
   587→      const url = new URL(v)
   588→      const s = url.searchParams.get('s')
   589→      const t = url.searchParams.get('t')
   590→      if (s && t) {
   591→        window.location.assign(`/?s=${encodeURIComponent(s)}&t=${encodeURIComponent(t)}`)
   592→        return
   593→      }
   594→    } catch {
   595→      // not a URL
   596→    }
   597→    const m = v.match(/^([A-HJ-NP-Z2-9]{6})-([A-Za-z0-9]{8,128})$/i)
   598→    if (m) {
   599→      window.location.assign(`/?s=${encodeURIComponent(m[1]!.toUpperCase())}&t=${encodeURIComponent(m[2]!)}`)
   600→      return
   601→    }
   602→    window.location.assign('/')
   603→  }
   604→
   605→  return (
   606→    <StateShell tone="border-destructive/30 bg-destructive/5">
   607→      <span className="flex h-14 w-14 items-center justify-center rounded-full bg-destructive/10">
   608→        <CloudOff className="h-6 w-6 text-destructive" aria-hidden />
   609→      </span>
   610→      <p className="mt-4 font-semibold">{error?.title ?? 'Invalid or expired link'}</p>
   611→      <p className="mt-1 text-sm text-muted-foreground">
   612→        {error?.message ?? 'This QR link is no longer valid. Scan a fresh code from the desktop.'}
   613→      </p>
   614→      <div className="mt-5 w-full space-y-2">
   615→        <Input
   616→          value={manual}
   617→          onChange={(e) => setManual(e.target.value)}
   618→          placeholder="Paste pairing link or CODE-KEY"
   619→          aria-label="Pairing link or manual code"
   620→          className="h-11 bg-background text-center"
   621→        />
   622→        <Button className="h-11 w-full" onClick={tryManual}>
   623→          <FilePlus2 className="h-4 w-4" aria-hidden />
   624→          Try pairing key
   625→        </Button>
   626→      </div>
   627→      <StartOverLink />
   628→    </StateShell>
   629→  )
   630→}
