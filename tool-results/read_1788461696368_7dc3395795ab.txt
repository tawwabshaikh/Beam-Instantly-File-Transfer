   174→  if (!socket || !socket.connected) return

   180→function patchTransfer(id: string, patch: Partial<TransferRow>): void {
   181→  useBeamStore.setState((state) => {
   182→    const row = state.transfers[id]
   183→    if (!row) return state
   184→    beamDebug.storeWrites++
   185→    return { transfers: { ...state.transfers, [id]: { ...row, ...patch } } }
   186→  })
   187→}

   189→function upsertRow(id: string, row: TransferRow): void {
   190→  useBeamStore.setState((state) => ({
   191→    transfers: { ...state.transfers, [id]: { ...state.transfers[id], ...row } },
   192→  }))
   193→}

   195→function noteProgress(id: string, transferred: number, size: number): void {
   196→  beamDebug.noteProgressCalls++
   197→  const now = Date.now()
   198→  const arr = samples.get(id) ?? []
   199→  arr.push({ t: now, b: transferred })
   200→  while (arr.length > 2 && now - arr[0].t > 4000) arr.shift()
   201→  samples.set(id, arr)
   202→
   203→  let speed = lastSpeed.get(id) ?? 0
   204→  const first = arr[0]
   205→  const last = arr[arr.length - 1]
   206→  if (arr.length >= 2 && last.t - first.t > 300) {
   207→    const inst = (last.b - first.b) / ((last.t - first.t) / 1000)
   208→    speed = speed === 0 ? inst : speed * 0.6 + inst * 0.4
   209→    lastSpeed.set(id, speed)
   210→  }
   211→
   212→  const lastFlush = flushTimes.get(id) ?? 0
   213→  if (now - lastFlush >= 120) {
   214→    flushTimes.set(id, now)
   215→    beamDebug.flushes++
   216→    patchTransfer(id, {
   217→      transferred,
   218→      speed,
   219→      etaSec: speed > 0 && transferred < size ? (size - transferred) / speed : null,
   220→    })
   221→  }
   222→}

   238→function handleStart(transferId: string, file: FileMeta, direction: TransferDirection): void {
   239→  canceledIds.delete(transferId)
   240→  sinks.set(transferId, { parts: [], received: 0, meta: file, direction })
   241→  samples.delete(transferId)
   242→  lastSpeed.delete(transferId)
   243→  upsertRow(transferId, {
   244→    id: transferId,
   245→    fileId: file.id,
   246→    name: file.name,
   247→    size: file.size,
   248→    type: file.type,
   249→    direction,
   250→    status: 'active',
   251→    transferred: 0,
   252→    speed: 0,
   253→    etaSec: null,
   254→    startedAt: Date.now(),
   255→    error: null,
   256→    transport: useBeamStore.getState().mode === 'webrtc' ? 'webrtc' : 'relay',
   257→  })
   258→}

   260→function handleChunk(transferId: string, data: ArrayBuffer): void {
   261→  beamDebug.chunksHandled++
   262→  const sink = sinks.get(transferId)
   263→  if (!sink || canceledIds.has(transferId)) return
   264→  sink.parts.push(data)
   265→  sink.received += data.byteLength
   266→  noteProgress(transferId, sink.received, sink.meta.size)
   267→  // Ack every chunk — cheap, and the relay sender uses it for flow control.
   268→  emitCode('beam:transfer:ack', { transferId, seq: 0 })
   269→}
   270→
   271→function handleDone(transferId: string): void {
   272→  const sink = sinks.get(transferId)
   273→  if (!sink) return
   274→  sinks.delete(transferId)
   275→  inboundDcTransfer = null
   276→  const blob = new Blob(sink.parts, { type: sink.meta.type || 'application/octet-stream' })
   277→  const url = URL.createObjectURL(blob)
   278→  const state = useBeamStore.getState()
   279→
   280→  const received: ReceivedFile = {
   281→    id: transferId,
   282→    name: sink.meta.name,
   283→    size: sink.meta.size,
   284→    type: sink.meta.type,
   285→    url,
   286→    direction: sink.direction,
   287→    receivedAt: Date.now(),
   288→  }
   289→
   290→  if (state.role === 'guest') triggerDownload(url, sink.meta.name)
   291→
   292→  useBeamStore.setState((s) => ({
   293→    transfers: {
   294→      ...s.transfers,
   295→      [transferId]: {
   296→        ...s.transfers[transferId],
   297→        status: 'done',
   298→        transferred: sink.meta.size,
   299→        speed: 0,
   300→        etaSec: 0,
   301→      },
   302→    },
   303→    received: [received, ...s.received.filter((r) => r.id !== transferId)],
   304→    stats: {
   305→      filesTransferred: s.stats.filesTransferred + 1,
   306→      totalData: s.stats.totalData + sink.meta.size,
   307→    },
   308→  }))
   309→
   310→  recordHistory({
   311→    name: sink.meta.name,
   312→    size: sink.meta.size,
   313→    direction: sink.direction,
   314→    status: 'completed',
   315→    sessionCode: state.session?.code ?? '',
   316→  })
   317→
   318→  notify(
   319→    'success',
   320→    state.role === 'guest' ? 'Download completed' : 'File received',
   321→    sink.meta.name,
   322→  )
   323→  playChime()
   324→  pumpSend()
   325→}

   361→
   362→function pumpSend(): void {
   363→  if (sendActive || sendQueue.length === 0) return
   364→  const next = sendQueue.shift()
   365→  if (!next) return
   366→  sendActive = true
   367→  const { transferId, fileId } = next
   368→  const file = fileObjects.get(fileId)
   369→  if (!file) {
   370→    emitCode('beam:transfer:error', { transferId, message: 'File is no longer available on this device' })
   371→    patchTransfer(transferId, { status: 'error', error: 'File no longer available' })
   372→    sendActive = false
   373→    pumpSend()
   374→    return
   375→  }
   376→  const meta: FileMeta = { id: fileId, name: file.name, size: file.size, type: file.type }
   377→  processSend(transferId, file, meta)
   378→    .catch((err) => {
   379→      const message = err instanceof Error ? err.message : 'Transfer failed'
   380→      emitCode('beam:transfer:error', { transferId, message })
   381→      patchTransfer(transferId, { status: 'error', error: message, speed: 0 })
   382→      notify('error', 'Transfer failed', `${file.name} — ${message}`)
   383→    })
   384→    .finally(() => {
   385→      sendActive = false
   386→      pumpSend()
   387→    })
   388→}

   390→async function processSend(transferId: string, file: File, meta: FileMeta): Promise<void> {
   391→  if (canceledIds.has(transferId)) {
   392→    patchTransfer(transferId, { status: 'canceled' })
   393→    return
   394→  }
   395→  patchTransfer(transferId, { status: 'active', startedAt: Date.now(), error: null })
   396→
   397→  const useDC = useBeamStore.getState().mode === 'webrtc' && getOutboundChannel()?.readyState === 'open'
   398→  if (useDC) {
   399→    patchTransfer(transferId, { transport: 'webrtc' })
   400→    try {
   401→      await sendOverDC(transferId, file, meta)
   402→      finishSend(transferId, file, meta)
   403→      return
   404→    } catch (err) {
   405→      if (canceledIds.has(transferId)) return patchTransfer(transferId, { status: 'canceled' })
   406→      // Data channel broke mid-transfer — one clean retry over the relay.
   407→      notify('info', 'Direct connection lost', 'Falling back to secure relay for this file…')
   408→    }
   409→  }
   410→
   411→  patchTransfer(transferId, { transport: 'relay' })
   412→  await sendOverRelay(transferId, file, meta)
   413→  finishSend(transferId, file, meta)
   414→}
   415→
   416→function finishSend(transferId: string, file: File, meta: FileMeta): void {
   417→  if (canceledIds.has(transferId)) {
   418→    patchTransfer(transferId, { status: 'canceled' })
   419→    return
   420→  }
   421→  const state = useBeamStore.getState()
   422→  useBeamStore.setState((s) => ({
   423→    transfers: {
   424→      ...s.transfers,
   425→      [transferId]: {
   426→        ...s.transfers[transferId],
   427→        status: 'done',
   428→        transferred: meta.size,
   429→        speed: 0,
   430→        etaSec: 0,
   431→      },
   432→    },
   433→    stats: {
   434→      filesTransferred: s.stats.filesTransferred + 1,
   435→      totalData: s.stats.totalData + meta.size,
   436→    },
   437→  }))
   438→  recordHistory({
   439→    name: meta.name,
   440→    size: meta.size,
   441→    direction: state.role === 'host' ? 'd2p' : 'p2d',
   442→    status: 'completed',
   443→    sessionCode: state.session?.code ?? '',
   444→  })
   445→  playChime()
   446→  notify('success', 'Transfer completed', file.name)
   447→}

   449→async function sendOverDC(transferId: string, file: File, meta: FileMeta): Promise<void> {
   450→  const dc = getOutboundChannel()
   451→  if (!dc || dc.readyState !== 'open') throw new Error('Direct channel not open')
   452→  dc.send(JSON.stringify({ kind: 'start', transferId, file: meta }))
   453→
   454→  const CHUNK = LIMITS.CHUNK_SIZE_WEBRTC
   455→  let offset = 0
   456→  while (offset < file.size) {
   457→    if (dc.readyState !== 'open') throw new Error('Connection lost during transfer')
   458→    if (canceledIds.has(transferId)) throw new CancelledError()
   459→    if (dc.bufferedAmount > LIMITS.BUFFER_HIGH_WATER) await waitForDrain(dc)
   460→    const buf = await file.slice(offset, offset + CHUNK).arrayBuffer()
   461→    dc.send(buf)
   462→    offset += buf.byteLength
   463→    noteProgress(transferId, offset, file.size)
   464→  }
   465→  dc.send(JSON.stringify({ kind: 'done', transferId }))
   466→}

   474→async function sendOverRelay(transferId: string, file: File, meta: FileMeta): Promise<void> {
   475→  emitCode('beam:transfer:start', { transferId, file: meta, direction: useBeamStore.getState().role === 'host' ? 'd2p' : 'p2d' })
   476→
   477→  const CHUNK = LIMITS.CHUNK_SIZE_RELAY
   478→  let offset = 0
   479→  relayInflight.set(transferId, 0)
   480→
   481→  while (offset < file.size) {
   482→    if (canceledIds.has(transferId)) throw new CancelledError()
   483→    if (!socket || !socket.connected) throw new Error('Connection lost during transfer')
   484→    if ((relayInflight.get(transferId) ?? 0) >= LIMITS.RELAY_WINDOW) {
   485→      await waitRelayBelow(transferId, LIMITS.RELAY_WINDOW)
   486→    }
   487→    const buf = await file.slice(offset, offset + CHUNK).arrayBuffer()
   488→    emitCode('beam:transfer:chunk', { transferId, seq: offset / CHUNK, data: buf })
   489→    offset += buf.byteLength
   490→    relayInflight.set(transferId, (relayInflight.get(transferId) ?? 0) + 1)
   491→    noteProgress(transferId, offset, file.size)
   492→  }
   493→
   494→  // Drain: wait until every sent chunk has been acknowledged. The unacked
   495→  // count is decremented by the ack handler even when no waiter is pending,
   496→  // so acks that arrived early are never "lost".
   497→  while ((relayInflight.get(transferId) ?? 0) > 0) {
   498→    await waitRelayBelow(transferId, 1)
   499→  }
   500→  relayInflight.delete(transferId)
   501→  emitCode('beam:transfer:done', { transferId })
   502→}
   503→
   504→/**
   505→ * Resolve when the unacked count drops below `threshold`. Each incoming ack
   506→ * wakes exactly one waiter; callers re-check their predicate after waking.
   507→ */
   508→function waitRelayBelow(transferId: string, threshold: number): Promise<void> {
   509→  // Predicate already satisfied (e.g. acks arrived before we started waiting)?
   510→  if ((relayInflight.get(transferId) ?? 0) < threshold) {
   511→    return Promise.resolve()
   512→  }
   513→  return new Promise((resolve, reject) => {
   514→    const entry = {
   515→      resolve,
   516→      timer: setTimeout(() => {
   517→        const q = ackWaiters.get(transferId)
   518→        if (q) ackWaiters.set(transferId, q.filter((e) => e !== entry))
   519→        reject(new Error('Transfer stalled — no acknowledgement from receiver'))
   520→      }, 30_000),
   521→    }
   522→    const q = ackWaiters.get(transferId) ?? []
   523→    q.push(entry)
   524→    ackWaiters.set(transferId, q)
   525→  })
   526→}
   527→
   528→function resolveAcks(transferId: string): void {
   529→  // Every ack decrements the true unacked count — even without a pending
   530→  // waiter — so early acks are never lost.
   531→  const cur = relayInflight.get(transferId)
   532→  if (cur !== undefined && cur > 0) relayInflight.set(transferId, cur - 1)
   533→
   534→  const q = ackWaiters.get(transferId)
   535→  const entry = q?.shift()
   536→  if (entry) {
   537→    clearTimeout(entry.timer)
   538→    entry.resolve()
   539→    if (q && q.length === 0) ackWaiters.delete(transferId)
   540→  }
   541→}

   551→function wireChannel(dc: RTCDataChannel, label: string): void {
   552→  dc.binaryType = 'arraybuffer'
   553→  dc.onmessage = (e) => {
   554→    try {
   555→      if (typeof e.data === 'string') {
   556→        const msg = JSON.parse(e.data) as {
   557→          kind: 'start' | 'done' | 'cancel'
   558→          transferId: string
   559→          file?: FileMeta
   560→        }
   561→        if (msg.kind === 'start' && msg.file) {
   562→          inboundDcTransfer = msg.transferId
   563→          const direction: TransferDirection = label === CHANNEL_D2P ? 'd2p' : 'p2d'
   564→          handleStart(msg.transferId, msg.file, direction)
   565→        } else if (msg.kind === 'done') {
   566→          handleDone(msg.transferId)
   567→        } else if (msg.kind === 'cancel') {
   568→          handleCancel(msg.transferId)
   569→        }
   570→      } else if (e.data instanceof ArrayBuffer) {
   571→        if (inboundDcTransfer) handleChunk(inboundDcTransfer, e.data)
   572→      } else if (e.data instanceof Blob) {
   573→        // Safari may deliver Blob unless binaryType was set before messages arrived
   574→        e.data.arrayBuffer().then((buf) => {
   575→          if (inboundDcTransfer) handleChunk(inboundDcTransfer, buf)
   576→        })
   577→      }
   578→    } catch {
   579→      // ignore malformed control frames
   580→    }
   581→  }
   582→  dc.onclose = () => {
   583→    if (label === CHANNEL_D2P) dcD2P = null
   584→    else dcP2D = null
   585→  }
   586→}

   588→function destroyPeer(): void {
   589→  if (p2pTimer) {
   590→    clearTimeout(p2pTimer)
   591→    p2pTimer = null
   592→  }
   593→  try {
   594→    pc?.close()
   595→  } catch {
   596→    // noop
   597→  }
   598→  pc = null
   599→  dcD2P = null
   600→  dcP2D = null
   601→  inboundDcTransfer = null
   602→}
   603→
   604→function switchMode(mode: TransportMode): void {
   605→  const st = useBeamStore.getState()
   606→  if (st.mode !== 'none' || st.phase === 'expired' || st.phase === 'ended') return
   607→  useBeamStore.setState({ mode, phase: 'connected', connectedAt: Date.now() })
   608→  emitCode('beam:mode', { mode })
   609→  if (p2pTimer) {
   610→    clearTimeout(p2pTimer)
   611→    p2pTimer = null
   612→  }
   613→  if (guestTimer) {
   614→    clearTimeout(guestTimer)
   615→    guestTimer = null
   616→  }
   617→}
   618→
   619→async function hostStartPeerConnection(): Promise<void> {
   620→  destroyPeer()
   621→  const state = useBeamStore.getState()
   622→  if (!state.session) return
   623→  useBeamStore.setState({ phase: 'connecting', mode: 'none' })
   624→
   625→  const peer = createPeerConnection()
   626→  pc = peer
   627→
   628→  const out = peer.createDataChannel(CHANNEL_D2P, { ordered: true })
   629→  dcD2P = out
   630→  wireChannel(out, CHANNEL_D2P)
   631→  out.onopen = () => {
   632→    // Both channels negotiated in the same SDP — outbound open ⇒ ready.
   633→    switchMode('webrtc')
   634→  }
   635→
   636→  peer.ondatachannel = (e) => {
   637→    if (e.channel.label === CHANNEL_P2D) {
   638→      dcP2D = e.channel
   639→      wireChannel(e.channel, CHANNEL_P2D)
   640→    }
   641→  }
   642→  peer.onicecandidate = (e) => {
   643→    if (e.candidate) emitCode('beam:signal', { signal: { type: 'candidate', payload: e.candidate.toJSON() } })
   644→  }
   645→  peer.onconnectionstatechange = () => {
   646→    if (peer.connectionState === 'failed' && useBeamStore.getState().mode === 'none') {
   647→      switchMode('relay')
   648→    }
   649→  }
   650→
   651→  const offer = await peer.createOffer()
   652→  await peer.setLocalDescription(offer)
   653→  emitCode('beam:signal', { signal: { type: 'offer', payload: peer.localDescription } })
   654→
   655→  p2pTimer = setTimeout(() => {
   656→    if (useBeamStore.getState().mode === 'none') switchMode('relay')
   657→  }, LIMITS.P2P_TIMEOUT_MS)
   658→}
   659→
   660→async function guestAcceptOffer(offer: RTCSessionDescriptionInit): Promise<void> {
   661→  destroyPeer()
   662→  const peer = createPeerConnection()
   663→  pc = peer
   664→  useBeamStore.setState({ phase: 'connecting' })
   665→
   666→  peer.ondatachannel = (e) => {
   667→    if (e.channel.label === CHANNEL_D2P) {
   668→      dcD2P = e.channel
   669→      wireChannel(e.channel, CHANNEL_D2P)
   670→    } else if (e.channel.label === CHANNEL_P2D) {
   671→      dcP2D = e.channel
   672→      wireChannel(e.channel, CHANNEL_P2D)
   673→    }
   674→  }
   675→  peer.onicecandidate = (e) => {
   676→    if (e.candidate) emitCode('beam:signal', { signal: { type: 'candidate', payload: e.candidate.toJSON() } })
   677→  }
   678→
   679→  await peer.setRemoteDescription(offer)
   680→  const answer = await peer.createAnswer()
   681→  await peer.setLocalDescription(answer)
   682→  emitCode('beam:signal', { signal: { type: 'answer', payload: peer.localDescription } })
   683→}

   689→function ensureSocket(): Socket {
   690→  if (socket) return socket
   691→  socket = io(SIGNALING_WS_URL, {
   692→    path: '/',
   693→    transports: ['websocket', 'polling'],
   694→    reconnection: true,
   695→    reconnectionAttempts: 10,
   696→    reconnectionDelay: 1000,
   697→    timeout: 10000,
   698→  })
   699→
   700→  socket.on('connect', () => {
   701→    joinAttempts = 0
   702→    emitJoin()
   703→  })
   704→  socket.on('beam:joined', (data: { ok: true; role: Role; session: { code: string; expiresAt: number; status: string; fileManifest: FileMeta[]; hostConnected: boolean; guestConnected: boolean; guestDevice: DeviceInfo | null } }) => {
   705→    const st = useBeamStore.getState()
   706→    if (!data?.ok || !st.session || data.role !== st.role) return
   707→    joiningCode = null
   708→    joinSentFor = null
   709→    useBeamStore.setState({ manifest: data.session.fileManifest ?? [] })
   710→
   711→    if (st.role === 'host') {
   712→      if (st.phase !== 'connected') useBeamStore.setState({ phase: 'waiting' })
   713→    } else {
   714→      if (st.phase === 'opening' || st.phase === 'lost') {
   715→        useBeamStore.setState({ phase: 'connecting' })
   716→        if (guestTimer) clearTimeout(guestTimer)
   717→        guestTimer = setTimeout(() => {
   718→          if (useBeamStore.getState().phase === 'connecting') {
   719→            useBeamStore.setState({
   720→              phase: 'lost',
   721→              error: { title: 'Connection timed out', message: 'Could not reach the desktop. Make sure the Beam tab is still open on your computer, then try again.' },
   722→            })
   723→          }
   724→        }, 30_000)
   725→      }
   726→    }
   727→  })
   728→
   729→  socket.on('beam:error', (err: { ok: false; code: string; message: string }) => {
   730→    joiningCode = null
   731→    joinSentFor = null
   732→    const st = useBeamStore.getState()
   733→    const message = err?.message ?? 'Unknown error'
   734→    if (err?.code === 'EXPIRED') {
   735→      useBeamStore.setState({ phase: 'expired', error: { title: 'Session expired', message } })
   736→    } else if (err?.code === 'NOT_FOUND' || err?.code === 'INVALID_TOKEN' || err?.code === 'ENDED') {
   737→      if (st.role === 'guest') {
   738→        useBeamStore.setState({ phase: 'invalid', error: { title: 'Invalid or expired link', message } })
   739→      } else {
   740→        useBeamStore.setState({ phase: 'failed', error: { title: 'Session error', message } })
   741→      }
   742→    } else if (err?.code === 'ROLE_TAKEN') {
   743→      useBeamStore.setState({ phase: 'failed', error: { title: 'Device already paired', message: 'Another device is already connected to this session.' } })
   744→    } else if (err?.code === 'RATE_LIMITED') {
   745→      useBeamStore.setState({ phase: 'failed', error: { title: 'Too many attempts', message } })
   746→    } else {
   747→      useBeamStore.setState({ phase: 'failed', error: { title: 'Connection error', message } })
   748→    }
   749→  })
   750→
   751→  socket.on('beam:peer', (data: { event: 'joined' | 'left'; device: DeviceInfo | null }) => {
   752→    const st = useBeamStore.getState()
   753→    if (!st.session) return
   754→    if (data.event === 'joined') {
   755→      useBeamStore.setState({ peerDevice: data.device })
   756→      if (st.role === 'host') {
   757→        if (data.device) notify('success', 'Phone connected', `${describeDevice(data.device)} is ready to exchange files`)
   758→        hostStartPeerConnection().catch(() => switchMode('relay'))
   759→      } else if (st.phase === 'lost' || st.phase === 'failed') {
   760→        useBeamStore.setState({ phase: 'connecting', error: null })
   761→        if (guestTimer) clearTimeout(guestTimer)
   762→        guestTimer = setTimeout(() => {
   763→          if (useBeamStore.getState().phase === 'connecting') {
   764→            useBeamStore.setState({ phase: 'lost', error: { title: 'Connection timed out', message: 'Could not reach the desktop. Try reconnecting.' } })
   765→          }
   766→        }, 30_000)
   767→      }
   768→    } else {
   769→      useBeamStore.setState({ peerDevice: null, mode: 'none', connectedAt: null })
   770→      destroyPeer()
   771→      if (st.role === 'host') {
   772→        useBeamStore.setState({ phase: 'waiting' })
   773→        // Fail in-flight transfers so they can be retried after re-pairing
   774→        Object.values(useBeamStore.getState().transfers).forEach((row) => {
   775→          if (row.status === 'active' || row.status === 'queued') {
   776→            patchTransfer(row.id, { status: 'error', error: 'Connection lost', speed: 0 })
   777→          }
   778→        })
   779→        sinks.clear()
   780→        ackWaiters.clear()
   781→        sendQueue = []
   782→        sendActive = false
   783→        notify('info', 'Phone disconnected', 'Waiting for a device to rejoin — the QR code is still valid.')
   784→      } else {
   785→        useBeamStore.setState({ phase: 'lost', error: { title: 'Connection lost', message: 'The desktop connection was interrupted. Reconnect to continue.' } })
   786→      }
   787→    }
   788→  })
   789→
   790→  socket.on('beam:mode', (data: { mode: TransportMode }) => {
   791→    if (useBeamStore.getState().role !== 'guest') return
   792→    if (data?.mode === 'webrtc' || data?.mode === 'relay') switchMode(data.mode)
   793→  })
   794→
   795→  socket.on('beam:manifest', (data: { files: FileMeta[] }) => {
   796→    if (useBeamStore.getState().role !== 'guest') return
   797→    if (Array.isArray(data?.files)) useBeamStore.setState({ manifest: data.files })
   798→  })
   799→
   800→  socket.on('beam:incoming', (data: { files: FileMeta[] }) => {
   801→    const st = useBeamStore.getState()
   802→    if (st.role !== 'host' || !Array.isArray(data?.files)) return
   803→    const existing = new Set(st.incomingFiles.map((f) => f.id))
   804→    const fresh = data.files.filter((f) => f && f.id && !existing.has(f.id) && f.size <= LIMITS.MAX_FILE_BYTES)
   805→    if (fresh.length === 0) return
   806→    useBeamStore.setState({ incomingFiles: [...st.incomingFiles, ...fresh] })
   807→    for (const file of fresh) {
   808→      const transferId = genId()
   809→      upsertRow(transferId, {
   810→        id: transferId,
   811→        fileId: file.id,
   812→        name: file.name,
   813→        size: file.size,
   814→        type: file.type,
   815→        direction: 'p2d',
   816→        status: 'queued',
   817→        transferred: 0,
   818→        speed: 0,
   819→        etaSec: null,
   820→        startedAt: null,
   821→        error: null,
   822→        transport: null,
   823→      })
   824→      emitCode('beam:transfer:request', { transferId, fileId: file.id, direction: 'p2d' })
   825→    }
   826→  })
   827→
   828→  socket.on('beam:transfer:request', (data: { transferId: string; fileId: string; direction: TransferDirection }) => {
   829→    const st = useBeamStore.getState()
   830→    if (!st.session || !data?.transferId || !data?.fileId) return
   831→    if (st.role === 'host' && data.direction !== 'd2p') return
   832→    if (st.role === 'guest' && data.direction !== 'p2d') return
   833→    const metaFile = st.role === 'host'
   834→      ? st.selectedFiles.find((f) => f.id === data.fileId)
   835→      : st.mobileFiles.find((f) => f.id === data.fileId)
   836→    upsertRow(data.transferId, {
   837→      id: data.transferId,
   838→      fileId: data.fileId,
   839→      name: metaFile?.name ?? 'File',
   840→      size: metaFile?.size ?? 0,
   841→      type: metaFile?.type ?? '',
   842→      direction: data.direction,
   843→      status: 'queued',
   844→      transferred: 0,
   845→      speed: 0,
   846→      etaSec: null,
   847→      startedAt: null,
   848→      error: null,
   849→      transport: null,
   850→    })
   851→    enqueueSend(data.transferId, data.fileId)
   852→  })
   853→
   854→  socket.on('beam:transfer:start', (data: { transferId: string; file: FileMeta; direction: TransferDirection }) => {
   855→    if (!data?.transferId || !data?.file) return
   856→    handleStart(data.transferId, data.file, data.direction)
   857→  })
   858→
   859→  socket.on('beam:transfer:chunk', (data: { transferId: string; data: ArrayBuffer }) => {
   860→    if (!data?.transferId || !data?.data) return
   861→    handleChunk(data.transferId, data.data)
   862→  })
   863→
   864→  socket.on('beam:transfer:ack', (data: { transferId: string }) => {
   865→    if (!data?.transferId) return
   866→    resolveAcks(data.transferId)
   867→  })
   868→
   869→  socket.on('beam:transfer:done', (data: { transferId: string }) => {
   870→    if (!data?.transferId) return
   871→    handleDone(data.transferId)
   872→  })
   873→
   874→  socket.on('beam:transfer:cancel', (data: { transferId: string }) => {
   875→    if (!data?.transferId) return
   876→    handleCancel(data.transferId)
   877→  })
   878→
   879→  socket.on('beam:transfer:error', (data: { transferId: string; message: string }) => {
   880→    if (!data?.transferId) return
   881→    handleTransferError(data.transferId, data.message ?? 'Transfer failed')
   882→  })
   883→
   884→  socket.on('beam:ended', () => {
   885→    destroyPeer()
   886→    useBeamStore.setState({ phase: 'ended', mode: 'none', peerDevice: null })
   887→    notify('info', 'Session ended', 'This transfer session has been closed.')
   888→  })
   889→
   890→  socket.on('beam:expired', () => {
   891→    destroyPeer()
   892→    useBeamStore.setState({ phase: 'expired', mode: 'none', peerDevice: null })
   893→  })
   894→
   895→  socket.on('disconnect', (reason: string) => {
   896→    const st = useBeamStore.getState()
   897→    if (st.phase === 'connected' || st.phase === 'connecting') {
   898→      destroyPeer()
   899→      useBeamStore.setState({ mode: 'none' })
   900→      if (st.role === 'guest') {
   901→        useBeamStore.setState({ phase: 'lost', error: { title: 'Connection lost', message: 'Your connection was interrupted. Reconnecting…' } })
   902→      }
   903→    }
   904→  })
   905→
   906→  return socket
   907→}

   909→function emitJoin(): void {
   910→  const st = useBeamStore.getState()
   911→  if (!st.session || !st.role || !socket) return
   912→  if (joinSentFor === st.session.code) return
   913→  joinSentFor = st.session.code
   914→  socket.emit('beam:join', {
   915→    code: st.session.code,
   916→    token: st.session.token,
   917→    role: st.role,
   918→    device: getDeviceInfo(),
   919→  })
   920→}

   943→export const useBeamStore = create<BeamStore>()((set, get) => ({
   944→  ...initialState,
   945→
   946→  /* ---------------- desktop: file selection ---------------- */
   947→
   948→  addFiles(files: FileList | File[]) {
   949→    const list = Array.from(files)
   950→    const { accepted, rejections } = validateFiles(list, get().selectedFiles.reduce((a, f) => a + f.size, 0))
   951→    for (const r of rejections) notify('error', 'Cannot add file', `${r.name} — ${r.reason}`)
   952→    if (accepted.length === 0) return
   953→
   954→    const newFiles: SelectedFile[] = accepted.map((file) => ({
   955→      id: genId(),
   956→      name: file.name,
   957→      size: file.size,
   958→      type: file.type,
   959→      file,
   960→      previewUrl: makePreviewUrl(file),
   961→    }))
   962→    for (const f of newFiles) fileObjects.set(f.id, f.file)
   963→
   964→    set((s) => ({ selectedFiles: [...s.selectedFiles, ...newFiles] }))
   965→
   966→    const st = get()
   967→    if (!st.session && st.phase !== 'creating') {
   968→      void get().createNewSession()
   969→    } else if (st.session && st.role === 'host' && ['waiting', 'connecting', 'connected'].includes(st.phase)) {
   970→      emitCode('beam:manifest', {
   971→        files: get().selectedFiles.map((f) => ({ id: f.id, name: f.name, size: f.size, type: f.type })),
   972→      })
   973→    }
   974→  },
   975→
   976→  removeSelectedFile(id: string) {
   977→    const file = get().selectedFiles.find((f) => f.id === id)
   978→    if (file?.previewUrl) URL.revokeObjectURL(file.previewUrl)
   979→    fileObjects.delete(id)
   980→    // cancel any queued/active transfer of this file
   981→    Object.values(get().transfers).forEach((row) => {
   982→      if (row.fileId === id && (row.status === 'queued' || row.status === 'active')) {
   983→        canceledIds.add(row.id)
   984→        emitCode('beam:transfer:cancel', { transferId: row.id })
   985→        patchTransfer(row.id, { status: 'canceled' })
   986→      }
   987→    })
   988→    set((s) => ({ selectedFiles: s.selectedFiles.filter((f) => f.id !== id) }))
   989→    const st = get()
   990→    if (st.session && st.role === 'host' && ['waiting', 'connecting', 'connected'].includes(st.phase)) {
   991→      emitCode('beam:manifest', {
   992→        files: get().selectedFiles.map((f) => ({ id: f.id, name: f.name, size: f.size, type: f.type })),
   993→      })
   994→    }
   995→  },
   996→
   997→  /* ---------------- desktop: session lifecycle ---------------- */
   998→
   999→  async createNewSession() {
  1000→    const files = get().selectedFiles
  1001→    if (files.length === 0) return
  1002→    if (get().phase === 'creating') return
  1003→
  1004→    destroyPeer()
  1005→    sendQueue = []
  1006→    sendActive = false
  1007→    set({ ...initialState, selectedFiles: files, role: 'host', phase: 'creating' })
  1008→
  1009→    try {
  1010→      const metas = files.map((f) => ({ id: f.id, name: f.name, size: f.size, type: f.type }))
  1011→      const res = await api.createSession(metas)
  1012→      const joinUrl =
  1013→        typeof window !== 'undefined'
  1014→          ? `${window.location.origin}/?s=${encodeURIComponent(res.code)}&t=${encodeURIComponent(res.token)}`
  1015→          : ''
  1016→      set({
  1017→        session: { code: res.code, token: res.token, expiresAt: res.expiresAt, joinUrl },
  1018→        role: 'host',
  1019→      })
  1020→      ensureSocket()
  1021→      if (socket?.connected) emitJoin()
  1022→    } catch (err) {
  1023→      const { title, message } = api.friendlyError(err)
  1024→      set({ phase: 'failed', error: { title, message } })
  1025→    }
  1026→  },
  1027→
  1028→  async endSession() {
  1029→    const st = get()
  1030→    if (st.session && st.role === 'host') {
  1031→      emitCode('beam:end', {})
  1032→      void api.endSession(st.session.code, st.session.token).catch(() => undefined)
  1033→    } else if (st.session) {
  1034→      emitCode('beam:end', {})
  1035→    }
  1036→    destroyPeer()
  1037→    socket?.disconnect()
  1038→    socket = null
  1039→    set({ phase: 'ended', mode: 'none', peerDevice: null, connectedAt: null })
  1040→  },
  1041→
  1042→  resetAll() {
  1043→    destroyPeer()
  1044→    socket?.disconnect()
  1045→    socket = null
  1046→    joiningCode = null
  1047→    joinSentFor = null
  1048→    sendQueue = []
  1049→    sendActive = false
  1050→    sinks.clear()
  1051→    ackWaiters.clear()
  1052→    canceledIds.clear()
  1053→    samples.clear()
  1054→    lastSpeed.clear()
  1055→    get().selectedFiles.forEach((f) => f.previewUrl && URL.revokeObjectURL(f.previewUrl))
  1056→    set({ ...initialState })
  1057→  },
  1058→
  1059→  /* ---------------- phone: session open ---------------- */
  1060→
  1061→  async openSession(code: string, token: string) {
  1062→    // Idempotence guard (StrictMode double-mount / re-render)
  1063→    if (joiningCode === code && ['opening', 'connecting'].includes(get().phase)) return
  1064→    if (get().role === 'guest' && get().session?.code === code && ['connected', 'connecting', 'lost'].includes(get().phase)) return
  1065→
  1066→    set({ ...initialState, role: 'guest', phase: 'opening', session: { code, token, expiresAt: 0, joinUrl: '' } })
  1067→    joiningCode = code
  1068→
  1069→    try {
  1070→      const info = await api.getSession(code, token)
  1071→      if (info.session.status !== 'active') {
  1072→        set({
  1073→          phase: info.session.status === 'expired' ? 'expired' : 'invalid',
  1074→          error: { title: 'Session unavailable', message: 'This session is no longer active.' },
  1075→        })
  1076→        joiningCode = null
  1077→        return
  1078→      }
  1079→      set({ manifest: info.session.fileManifest ?? [] })
  1080→      ensureSocket()
  1081→      if (socket?.connected) emitJoin()
  1082→    } catch (err) {
  1083→      const { title, message } = api.friendlyError(err)
  1084→      const isExpiry = title === 'Session not found' || title === 'Session expired'
  1085→      set({
  1086→        phase: isExpiry ? 'invalid' : 'failed',
  1087→        error: { title, message },
  1088→      })
  1089→      joiningCode = null
  1090→    }
  1091→  },
  1092→
  1093→  reconnect() {
  1094→    const st = get()
  1095→    if (st.role !== 'guest' || !st.session) return
  1096→    joiningCode = null
  1097→    joinSentFor = null
  1098→    set({ phase: 'opening', error: null })
  1099→    ensureSocket()
  1100→    if (!socket?.connected) {
  1101→      // socket.io will auto-reconnect and emitJoin on 'connect'
  1102→      socket?.connect()
  1103→    } else {
  1104→      emitJoin()
  1105→    }
  1106→    set({ phase: 'connecting' })
  1107→  },
  1108→
  1109→  /* ---------------- phone: download ---------------- */
  1110→
  1111→  requestDownload(fileId: string) {
  1112→    const st = get()
  1113→    if (st.phase !== 'connected') {
  1114→      notify('info', 'Not connected', 'Wait until the connection is ready, then try again.')
  1115→      return
  1116→    }
  1117→    const file = st.manifest.find((f) => f.id === fileId)
  1118→    if (!file) return
  1119→    const transferId = genId()
  1120→    upsertRow(transferId, {
  1121→      id: transferId,

  1123→      name: file.name,

  1125→      type: file.type,
  1126→      direction: 'd2p',
  1127→      status: 'queued',
  1128→      transferred: 0,
  1129→      speed: 0,
  1130→      etaSec: null,
  1131→      startedAt: null,
  1132→      error: null,
  1133→      transport: null,
  1134→    })
  1135→    emitCode('beam:transfer:request', { transferId, fileId, direction: 'd2p' })
  1136→  },

  1138→  downloadAll() {

  1140→    const done = new Set(
  1141→      Object.values(st.transfers)
  1142→        .filter((t) => t.direction === 'd2p' && t.status === 'done')
  1143→        .map((t) => t.fileId),
  1144→    )
  1145→    const pending = st.manifest.filter((f) => !done.has(f.id))
  1146→    if (pending.length === 0) {

  1148→      return
  1149→    }
  1150→    // Small stagger keeps the send queue orderly
  1151→    pending.forEach((f, i) => setTimeout(() => get().requestDownload(f.id), i * 400))

  1156→  addMobileFiles(files: FileList | File[]) {

  1164→      name: file.name,

  1171→    set((s) => ({ mobileFiles: [...s.mobileFiles, ...newFiles] }))
  1172→
  1173→    const st = get()
  1174→    if (st.session && st.role === 'guest' && ['connected', 'connecting'].includes(st.phase)) {
  1175→      emitCode('beam:incoming', {
  1176→        files: st.mobileFiles.map((f) => ({ id: f.id, name: f.name, size: f.size, type: f.type })),
  1177→      })
  1178→    }

  1184→    fileObjects.delete(id)

  1187→        canceledIds.add(row.id)

  1193→    const st = get()
  1194→    if (st.session && st.role === 'guest' && ['connected', 'connecting'].includes(st.phase)) {
  1195→      emitCode('beam:incoming', {

  1197→      })

  1199→  },

  1210→  async shareReceived(id: string) {
  1211→    const file = get().received.find((r) => r.id === id)
  1212→    if (!file) return
  1213→    try {
  1214→      const blob = await fetch(file.url).then((r) => r.blob())
  1215→      const shareFile = new File([blob], file.name, { type: file.type || blob.type })
  1216→      const nav = navigator as Navigator & { canShare?: (data: ShareData) => boolean }
  1217→      if (nav.share && nav.canShare?.({ files: [shareFile] })) {
  1218→        await nav.share({ files: [shareFile], title: file.name })
  1219→        return
  1220→      }
  1221→      triggerDownload(file.url, file.name)
  1222→    } catch {
  1223→      // user cancelled or unsupported
  1224→    }
  1225→  },
  1226→
  1227→  dismissReceived(id: string) {

  1230→    set((s) => ({ received: s.received.filter((r) => r.id !== id) }))
  1231→  },

  1233→  retryTransfer(transferId: string) {

  1239→    patchTransfer(transferId, { status: 'canceled' })
  1240→    emitCode('beam:transfer:request', { transferId: newId, fileId: row.fileId, direction: row.direction })
  1241→  },

  1245→    emitCode('beam:transfer:cancel', { transferId })
