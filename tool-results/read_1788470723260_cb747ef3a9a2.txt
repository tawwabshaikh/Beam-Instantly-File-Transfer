    58→        {phase === 'opening' && <OpeningView />}
    59→        {phase === 'connecting' && <ConnectingView />}

    61→          phase === 'waiting' ||
    62→          phase === 'creating' ||

   318→/** Aggregate progress for a batch of downloads (e.g. "Download All"). */
   319→function BatchProgress({ rows }: { rows: TransferRow[] }) {
   320→  const relevant = rows.filter((r) => r.status !== 'canceled')
   321→  const total = relevant.reduce((a, r) => a + r.size, 0)
   322→  const done = relevant.reduce((a, r) => a + r.transferred, 0)
   323→  const active = relevant.filter((r) => r.status === 'active' || r.status === 'queued').length
   324→  const finished = relevant.filter((r) => r.status === 'done').length
   325→  const pct = total > 0 ? Math.min(100, (done / total) * 100) : 0
   326→
   327→  if (relevant.length < 2) return null
   328→
   329→  return (
   330→    <div className="bg-muted/40 px-4 py-2.5">
   331→      <div className="flex items-center justify-between text-xs">
   332→        <span className="font-medium text-foreground">
   333→          {active > 0 ? `Downloading ${finished + 1} of ${relevant.length}` : `${finished} of ${relevant.length} downloaded`}
   334→        </span>
   335→        <span className="tnum text-muted-foreground">
   336→          {formatBytes(done)} / {formatBytes(total)}
   337→        </span>
   338→      </div>
   339→      <Progress value={pct} className="mt-1.5 h-1.5" aria-label={`Batch progress ${pct.toFixed(0)}%`} />
   340→    </div>
   341→  )
   342→}

   344→function MobileFileRow({ fileId, flash = false }: { fileId: string; flash?: boolean }) {
   345→  const file = useBeamStore((s) => s.manifest.find((f) => f.id === fileId))
   346→  const transfers = useBeamStore((s) => s.transfers)
   347→  const requestDownload = useBeamStore((s) => s.requestDownload)
   348→  const shareReceived = useBeamStore((s) => s.shareReceived)
   349→  const retryTransfer = useBeamStore((s) => s.retryTransfer)
   350→
   351→  const rows = useMemo(
   352→    () =>
   353→      Object.values(transfers).filter((t) => t.fileId === fileId && t.direction === 'd2p'),
   354→    [transfers, fileId],
   355→  )
   356→
   357→  if (!file) return null
   358→  const row = rows[rows.length - 1]
   359→  const busy = row && (row.status === 'queued' || row.status === 'active')
   360→  const failed = row?.status === 'error'
   361→  const done = rows.some((r) => r.status === 'done')
   362→  // Once downloaded, images get a real thumbnail from the received blob.
   363→  const receivedThumb = done && file.type.startsWith('image/')
   364→    ? useBeamStore.getState().received.find((r) => r.name === file.name && r.size === file.size)?.url ?? null
   365→    : null
   366→
   367→  return (
   368→    <li className={cn('flex items-center gap-3 px-4 py-3 transition-colors hover:bg-accent/40', flash && 'animate-flash')}>
   369→      {receivedThumb ? (
   370→        <img
   371→          src={receivedThumb}
   372→          alt=""
   373→          className="h-10 w-10 shrink-0 rounded-lg border border-border object-cover"
   374→          loading="lazy"
   375→        />
   376→      ) : (
   377→        <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-muted">
   378→          <FileTypeIcon type={file.type} name={file.name} className="h-4.5 w-4.5" />
   379→        </span>
   380→      )}
   381→      <div className="min-w-0 flex-1">
   382→        <p className="truncate text-sm font-medium" title={file.name}>
   383→          {file.name}
   384→        </p>
   385→        <p className={cn('tnum truncate text-xs', failed ? 'text-destructive' : 'text-muted-foreground')}>
   386→          {row?.status === 'active'
   387→            ? `${(row.size > 0 ? (row.transferred / row.size) * 100 : 0).toFixed(0)}% · ${formatSpeed(row.speed)}`
   388→            : failed
   389→              ? (row?.error ?? 'Download failed')
   390→              : formatBytes(file.size)}
   391→        </p>
   392→      </div>
   393→      {done ? (
   394→        <div className="flex items-center gap-1">
   395→          <Button
   396→            variant="ghost"
   397→            size="icon"
   398→            className="h-8 w-8"
   399→            onClick={() => {
   400→              const rec = useBeamStore.getState().received.find((r) => r.name === file.name)
   401→              if (rec) void shareReceived(rec.id)
   402→            }}
   403→            aria-label={`Share or re-save ${file.name}`}
   404→          >
   405→            <Share2 className="h-4 w-4" />
   406→          </Button>
   407→          <span className="inline-flex items-center gap-1 text-xs font-medium text-primary">
   408→            <CheckCircle2 className="h-4 w-4" aria-hidden />
   409→            Saved
   410→          </span>
   411→        </div>
   412→      ) : failed ? (
   413→        <Button
   414→          size="sm"
   415→          variant="outline"
   416→          className="h-9 min-w-20 border-destructive/40 text-destructive hover:bg-destructive/10 hover:text-destructive"
   417→          onClick={() => retryTransfer(row!.id)}
   418→        >
   419→          <RotateCcw className="h-4 w-4" aria-hidden />
   420→          Retry
   421→        </Button>
   422→      ) : (
   423→        <Button size="sm" className="h-9 min-w-24" onClick={() => requestDownload(fileId)} disabled={busy}>
   424→          {busy ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Download className="h-4 w-4" aria-hidden />}
   425→          {busy ? 'Busy…' : 'Download'}
   426→        </Button>
   427→      )}
   428→    </li>
   429→  )
   430→}

   514→                  <p className="truncate text-sm font-medium">{f.name}</p>
   515→                  <p className="tnum text-xs text-muted-foreground">
   516→                    {row?.status === 'active'
   517→                      ? `${(row.size > 0 ? (row.transferred / row.size) * 100 : 0).toFixed(0)}% · ${formatSpeed(row.speed)}`
   518→                      : row?.status === 'done'
   519→                        ? 'Delivered to desktop'
   520→                        : row?.status === 'queued'
   521→                          ? 'Queued…'
   522→                          : row?.status === 'error'
   523→                            ? row.error ?? 'Failed'
   524→                            : formatBytes(f.size)}
   525→                  </p>
   526→                </div>
   527→                {row?.status === 'done' ? (
   528→                  <CheckCircle2 className="h-5 w-5 shrink-0 text-primary" aria-label="Upload completed" />
   529→                ) : row?.status === 'error' ? (
   530→                  <Button
   531→                    variant="outline"
   532→                    size="sm"
   533→                    className="h-8 shrink-0 border-destructive/40 text-destructive hover:bg-destructive/10 hover:text-destructive"
   534→                    onClick={() => retryTransfer(row.id)}
   535→                  >
   536→                    <RotateCcw className="h-3.5 w-3.5" aria-hidden />
   537→                    Retry
   538→                  </Button>
   539→                ) : (
   540→                  <Button
   541→                    variant="ghost"
   542→                    size="icon"
   543→                    className="h-8 w-8 shrink-0 text-muted-foreground"
   544→                    onClick={() => removeMobileFile(f.id)}
   545→                    disabled={row?.status === 'active'}
   546→                    aria-label={`Stop sending ${f.name}`}
   547→                  >
   548→                    <X className="h-4 w-4" />
   549→                  </Button>
   550→                )}
   551→              </li>
   552→            )
   553→          })}
   554→        </ul>
   555→      )}
   556→
   557→      {active && (
   558→        <p className="tnum border-t border-border/70 px-4 py-2 text-center text-xs text-muted-foreground">
   559→          Uploading to desktop…
   560→        </p>
   561→      )}
   562→    </section>
   563→  )
   564→}

   606→function InvalidView() {
   607→  const error = useBeamStore((s) => s.error)
   608→  const [manual, setManual] = useState('')
   609→  const openSession = useBeamStore((s) => s.openSession)
   610→
   611→  const tryManual = () => {
   612→    const v = manual.trim()
   613→    try {
   614→      const url = new URL(v)
   615→      const s = url.searchParams.get('s')
   616→      const t = url.searchParams.get('t')
   617→      if (s && t) {
   618→        window.location.assign(`/?s=${encodeURIComponent(s)}&t=${encodeURIComponent(t)}`)
   619→        return
   620→      }
   621→    } catch {
   622→      // not a URL
   623→    }
   624→    const m = v.match(/^([A-HJ-NP-Z2-9]{6})-([A-Za-z0-9]{8,128})$/i)
   625→    if (m) {
   626→      window.location.assign(`/?s=${encodeURIComponent(m[1]!.toUpperCase())}&t=${encodeURIComponent(m[2]!)}`)
   627→      return
   628→    }
   629→    window.location.assign('/')
   630→  }
   631→
   632→  return (
   633→    <StateShell tone="border-destructive/30 bg-destructive/5">
   634→      <span className="flex h-14 w-14 items-center justify-center rounded-full bg-destructive/10">
   635→        <CloudOff className="h-6 w-6 text-destructive" aria-hidden />
   636→      </span>
   637→      <p className="mt-4 font-semibold">{error?.title ?? 'Invalid or expired link'}</p>
   638→      <p className="mt-1 text-sm text-muted-foreground">
   639→        {error?.message ?? 'This QR link is no longer valid. Scan a fresh code from the desktop.'}
   640→      </p>
   641→      <div className="mt-5 w-full space-y-2">
   642→        <Input
   643→          value={manual}
   644→          onChange={(e) => setManual(e.target.value)}
   645→          placeholder="Paste pairing link or CODE-KEY"
   646→          aria-label="Pairing link or manual code"
   647→          className="h-11 bg-background text-center"
   648→        />
   649→        <Button className="h-11 w-full" onClick={tryManual}>
   650→          <FilePlus2 className="h-4 w-4" aria-hidden />
   651→          Try pairing key
   652→        </Button>
   653→      </div>
   654→      <StartOverLink />
   655→    </StateShell>
   656→  )
   657→}
